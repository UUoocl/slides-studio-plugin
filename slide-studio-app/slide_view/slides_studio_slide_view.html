const urlParams = new URLSearchParams(window.location.search); const componentId = urlParams.get('id') || 'slideComponent'; const broadcastChannels = {}; (async () => { while (!window.ssSocket) { await new Promise(r => setTimeout(r, 100)); } // Connection listener loop (async () => { for await (let event of window.ssSocket.listener('connect')) { // Connected } })(); // Disconnection listener loop (async () => { for await (let event of window.ssSocket.listener('disconnect')) { // Disconnected } })(); // Subscribe specifically to the choreography command channel const cmdChannel = window.ssSocket.subscribe('custom_slidesCommands'); // Listener for modern Choreography updates (async () => { for await (let data of cmdChannel) { if (data.eventName === 'CHOREOGRAPHY_UPDATE') { const { config, transitions } = data.msgParam; console.log(`[SlideView] Choreography Update for ${componentId}:`, config); let myConfig = null; if (config.components) { myConfig = config.components.find(c => c.id === componentId); } else if (componentId === 'slideComponent') { myConfig = config.slideComponent; } if (myConfig) { const myTransition = (transitions || []).find(t => (t.sources || []).includes(componentId)); applySlideComponentConfig(myConfig, myTransition); } // Also check for legacy cameraComponent broadcast if we are the main slide view if (config.cameraComponent && componentId === 'slideComponent' && broadcastChannels.cameraShapes) { broadcastChannels.cameraShapes.postMessage(config.cameraComponent); } // Broadcast mask updates for components of type 'mask' if (config.components && broadcastChannels.cameraShapes) { config.components.forEach(c => { if (c.type === 'mask') { broadcastChannels.cameraShapes.postMessage(c); } }); } } } })(); })(); window.addEventListener('DOMContentLoaded', () => { broadcastChannels.cameraShapes = new BroadcastChannel("cameraShapes_channel"); }); function applySlideComponentConfig(config, transition) { const iframe = document.getElementById("revealIframe"); if (!iframe) return; // Apply direct transform and scaling const { x, y, scaleX, scaleY, style, transform } = config; // Support both old {x, y, scaleX, scaleY} and new {transform: {x, y, scale, rotation}} const finalX = transform?.x ?? x ?? 0; const finalY = transform?.y ?? y ?? 0; const finalScaleX = transform?.scale ?? scaleX ?? 1; const finalScaleY = transform?.scale ?? scaleY ?? 1; const finalRotation = transform?.rotation ?? 0; // Apply smooth transition if specified if (transition) { const { duration = 500, ease = 'ease-out' } = transition; // Map custom ease names to CSS equivalents let cssEase = ease; if (ease === 'bounce') cssEase = 'cubic-bezier(0.175, 0.885, 0.32, 1.275)'; if (ease === 'linear') cssEase = 'linear'; iframe.style.transition = `all ${duration}ms ${cssEase}`; } else { iframe.style.transition = "none"; } iframe.style.left = `${finalX}px`; iframe.style.top = `${finalY}px`; iframe.style.width = "100%"; iframe.style.height = "100%"; iframe.style.transform = `scale(${finalScaleX}, ${finalScaleY}) rotate(${finalRotation}deg)`; iframe.style.transformOrigin = "top left"; // Apply arbitrary CSS styles (filters, borders, etc.) if (style && typeof style === 'object') { Object.assign(iframe.style, style); } iframe.className = "slide-position"; }
